- No inventar features nuevas

### 2. Backend Middleware - Feature Gating

```typescript
// middleware/subscription.ts

import { Request, Response, NextFunction } from 'express';

// Verificar si el negocio tiene acceso a una feature
export const requireFeature = (feature: string) => {
  return async (req: Request, res: Response, next: NextFunction) => {
    try {
      const businessId = req.businessId; // from JWT
      const subscription = await getActiveSubscription(businessId);
      
      if (!subscription) {
        return res.status(402).json({
          error: 'Sin suscripción activa',
          requiresSubscription: true,
          suggestedPlan: 'florecer'
        });
      }
      
      const plan = await getPlan(subscription.planId);
      const hasFeature = plan.features[feature] === true || 
                         plan.features[feature] !== false; // para strings como "basic", "full"
      
      if (!hasFeature) {
        return res.status(403).json({
          error: `Feature "${feature}" no disponible en tu plan`,
          requiresUpgrade: true,
          currentPlan: plan.name_short,
          suggestedPlan: getSuggestedPlanForFeature(feature),
          upgradeUrl: '/subscription/upgrade'
        });
      }
      
      next();
    } catch (error) {
      res.status(500).json({ error: 'Error verificando suscripción' });
    }
  };
};

// Verificar límites de cantidad
export const checkLimit = (limitType: 'users' | 'products' | 'orders' | 'categories') => {
  return async (req: Request, res: Response, next: NextFunction) => {
    try {
      const businessId = req.businessId;
      const subscription = await getActiveSubscription(businessId);
      const plan = await getPlan(subscription.planId);
      
      const limitKey = `max_${limitType}`;
      const limit = plan[limitKey];
      
      if (!limit) {
        return next(); // NULL = ilimitado
      }
      
      // Contar cantidad actual
      const currentCount = await getCount(businessId, limitType);
      
      if (currentCount >= limit) {
        return res.status(429).json({
          error: `Límite alcanzado: ${currentCount}/${limit} ${limitType}`,
          limitReached: true,
          limit,
          current: currentCount,
          suggestedPlan: getSuggestedPlanForLimit(limitType, limit),
          upgradeUrl: '/subscription/upgrade'
        });
      }
      
      next();
    } catch (error) {
      res.status(500).json({ error: 'Error verificando límite' });
    }
  };
};

// Resetear contador de pedidos mensual
export const resetMonthlyOrderCount = async () => {
  const subscriptions = await getActiveSubscriptions();
  
  for (const sub of subscriptions) {
    const shouldReset = sub.last_order_count_reset && 
                        isCurrentMonth(sub.last_order_count_reset);
    
    if (shouldReset) {
      await updateSubscription(sub.id, {
        orders_this_month: 0,
        last_order_count_reset: new Date()
      });
    }
  }
};

// Helper: qué plan necesita para una feature

    'barcode': 'florecer',
    'calendar_view': 'florecer',

    'packages': 'crecimiento',
    'logistics': 'crecimiento',

    'afip_integration': 'crecimiento',
```

### 4. Frontend - Verificar Límites

```typescript
// hooks/useSubscriptionLimits.ts
export const useSubscriptionLimits = () => {
  const { subscription, plan } = useSubscription();
  
  const checkProductLimit = async () => {
    if (!plan.max_products) return { allowed: true }; // ilimitado
    
    const count = await fetchProductCount();
    return {
      allowed: count < plan.max_products,
      current: count,
      limit: plan.max_products,
      percentage: (count / plan.max_products) * 100
    };
  };
  
  const checkOrderLimit = async () => {
    if (!plan.max_orders_per_month) return { allowed: true };
    
    return {
      allowed: subscription.orders_this_month < plan.max_orders_per_month,
      current: subscription.orders_this_month,
      limit: plan.max_orders_per_month,
      percentage: (subscription.orders_this_month / plan.max_orders_per_month) * 100
    };
  };
  
  return { checkProductLimit, checkOrderLimit, subscription, plan };
};

  const [limit, setLimit] = useState(null);
  
  useEffect(() => {
    const check = type === 'products' ? checkProductLimit : checkOrderLimit;
    check().then(setLimit);
  }, [type]);
  
  if (!limit || limit.percentage < 80) return null;
  
  const isOver = limit.percentage >= 100;
  
  return (
    <Alert variant={isOver ? 'error' : 'warning'}>
      <AlertTitle>
        {isOver ? '❌ Límite alcanzado' : '⚠️ Casi llegás al límite'}
      </AlertTitle>
      <AlertDescription>
        {limit.current}/{limit.limit} {type} usados este mes
        {isOver && (
          <Button asChild size="sm" className="mt-2">
            <Link to="/subscription/upgrade">
              Cambiar de plan →
            </Link>
          </Button>
        )}
      </AlertDescription>
    </Alert>
  );
};
```

## 🎯 Recomendaciones Finales

### ✅ Hacer
- Precios founder bajos para captar mercado rápido ($12k/$24k)
- Lock de precio 12 meses para early adopters
- Onboarding gratis incluido en planes pagos
- 14 días de prueba gratis en planes pagos
- Hacer el plan del medio OBVIAMENTE el mejor
- Usar MercadoPago (es el estándar en Argentina)
- Incluir facturación AFIP como feature diferenciadora
- Mostrar límites de uso en la app ("25/30 pedidos usados")

### ❌ NO Hacer
- No cobrar setup fee (fricciona)
- No hacer plan gratis demasiado generoso
- No esconder precios
- No hacer contratos largos
- NO fijar precios en USD
- No ignorar inflación (revisar trimestralmente)
- No agregar features de más (usar lo que ya existe)

### 📌 Manejo de Inflación

**Opción recomendada: Ajuste Trimestral**
- Cada 3 meses, ajustar precios para NUEVOS clientes
- Clientes activos mantienen precio por 6 meses
- Comunicar con 30 días de anticipación
- Mensaje: "Ajuste por condiciones económicas"

**Alternativa: Incentivar Pago Anual**
- "Pagá $180,000 hoy y congelá el precio 12 meses"
- Cash flow inmediato + al cliente le conviene
- Win-win
